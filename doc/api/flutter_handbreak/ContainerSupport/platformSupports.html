static bool platformSupports(String platform, String container) {
  switch (platform) {
    case 'android':
      return container == 'mp4' || container == 'webm' || container == '3gp';
    case 'ios':
      return container == 'mp4' || container == 'mov';
    default:
      return container == 'mp4';
  }
}
