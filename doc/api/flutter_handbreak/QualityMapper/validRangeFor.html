Valid inclusive range for the numeric CQ value per codec. These mirror the encoder presets: x264 0..51, x265 0..51, VP9 0..63, AV1 0..63, MediaCodec QP 0..51 etc.

Implementation

static ({double min, double max}) validRangeFor(VideoCodec codec) {
  switch (codec) {
    case VideoCodec.h264:
      return (min: 0, max: 51); // x264 CRF 0(lossless)..51(worst)
    case VideoCodec.h265:
      return (min: 0, max: 51); // x265 CRF
    case VideoCodec.av1:
      return (min: 0, max: 63); // AV1 CQ / libaom cq-level
    case VideoCodec.vp9:
      return (min: 0, max: 63); // VP9 CQ
  }
}
